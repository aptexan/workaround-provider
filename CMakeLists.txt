# cmake build file for C++ tofu-plugin example.
# Assumes protobuf and gRPC have been installed using cmake.
# See cmake_externalproject/CMakeLists.txt for all-in-one cmake build
# that automatically builds all the dependencies before building ${proto_stem}.

cmake_minimum_required(VERSION 3.22)

project(tofu-plugin-cpp C CXX)

# --- Never search the Windows side of WSL ------------------------------------------------------------------------------
# Under WSL the Windows PATH is inherited (Anaconda, Visual Studio, ...) and CMake treats every PATH entry as a place to
# look for packages. It can then find a Windows Protobuf or Abseil and mix it with the Linux gRPC, which ends in hundreds
# of "Protobuf C++ gencode is built with an incompatible version" compile errors. So ignore every mounted Windows drive:
# /mnt/c always, and any other drive (D:, E:, ...) that /proc/mounts lists as a drvfs or 9p mount. Harmless elsewhere.
list(APPEND CMAKE_IGNORE_PREFIX_PATH /mnt/c)
if(EXISTS /proc/mounts)
  file(STRINGS /proc/mounts _mounts)
  foreach(_line IN LISTS _mounts)
    if(_line MATCHES "^[^ ]+ (/mnt/[^ ]+) (drvfs|9p) ")
      list(APPEND CMAKE_IGNORE_PREFIX_PATH "${CMAKE_MATCH_1}")
    endif()
  endforeach()
endif()
list(REMOVE_DUPLICATES CMAKE_IGNORE_PREFIX_PATH)
# CMAKE_IGNORE_PREFIX_PATH alone is not enough: find_package() still searches the directories of the PATH variable, and
# that is how a Windows Anaconda was found. So also remove every PATH entry under an ignored prefix, for this configure.
# (This changes only the environment of CMake itself; the shell you started it from is not touched.)
if(DEFINED ENV{PATH})
  string(REPLACE ":" ";" _path_entries "$ENV{PATH}")
  set(_kept_entries "")
  foreach(_entry IN LISTS _path_entries)
    set(_windows FALSE)
    foreach(_ignored IN LISTS CMAKE_IGNORE_PREFIX_PATH)
      string(FIND "${_entry}/" "${_ignored}/" _at)
      if(_at EQUAL 0)
        set(_windows TRUE)
      endif()
    endforeach()
    if(NOT _windows)
      list(APPEND _kept_entries "${_entry}")
    endif()
  endforeach()
  string(REPLACE ";" ":" _clean_path "${_kept_entries}")
  set(ENV{PATH} "${_clean_path}")
endif()

# --- Where gRPC is: the project's standard prefix -------------------------------------------------------------------
# gRPC, built against OpenSSL, is installed in ~/.local/grpc-openssl (BUILDING.md, docs/setup.md section 4.1). It is used
# when no CMAKE_PREFIX_PATH is given, so a build from scratch needs no -D option. This also matters because CMake can
# lose CMAKE_PREFIX_PATH by itself: when the compiler it recorded differs from the one a tool asks for (a command-line
# configure records /usr/bin/cc, VS Code's CMake Tools passes /usr/bin/gcc), it deletes the cache and re-runs the
# configure WITHOUT the -D options. With this, the order of command-line and VS Code configures does not matter.
# gRPC somewhere else: pass -DCMAKE_PREFIX_PATH=<its folder>.
if(NOT CMAKE_PREFIX_PATH AND EXISTS "$ENV{HOME}/.local/grpc-openssl")
  list(APPEND CMAKE_PREFIX_PATH "$ENV{HOME}/.local/grpc-openssl")
  message(STATUS "CMAKE_PREFIX_PATH was not given: using the standard prefix $ENV{HOME}/.local/grpc-openssl")
endif()

include(cmake/common.cmake)

# --- The dependencies must all come from ONE installation -----------------------------------------------------------
# Code generated by one Protobuf must be compiled against the headers of the same Protobuf, and gRPC must match both. A
# CMake cache that remembers packages from two places (for example after a configure without CMAKE_PREFIX_PATH) gives
# hundreds of confusing compile errors. Stop here instead, with the places listed. Skip with
# -DALLOW_MIXED_DEPENDENCY_PREFIXES=ON if you really mean to combine installations.
if(NOT ALLOW_MIXED_DEPENDENCY_PREFIXES)
  set(_prefixes "")
  set(_found "")
  foreach(_package IN ITEMS gRPC Protobuf absl utf8_range re2 c-ares)
    if(DEFINED ${_package}_DIR AND ${_package}_DIR)
      # .../lib/cmake/<package>  or  .../lib/<arch>/cmake/<package>   ->   the installation prefix
      string(REGEX REPLACE "/lib[^/]*(/[^/]+)?/cmake/[^/]+/?$" "" _prefix "${${_package}_DIR}")
      list(APPEND _prefixes "${_prefix}")
      string(APPEND _found "\n    ${_package}: ${${_package}_DIR}")
      foreach(_ignored IN LISTS CMAKE_IGNORE_PREFIX_PATH)
        string(FIND "${_prefix}/" "${_ignored}/" _at)
        if(_at EQUAL 0)
          message(FATAL_ERROR "${_package} was found at ${${_package}_DIR}, which is a Windows drive mounted in WSL "
                  "(ignored prefix ${_ignored}). Delete build/CMakeCache.txt and configure again with "
                  "-DCMAKE_PREFIX_PATH=<your gRPC prefix>.")
        endif()
      endforeach()
    endif()
  endforeach()
  list(REMOVE_DUPLICATES _prefixes)
  list(LENGTH _prefixes _prefix_count)
  if(_prefix_count GREATER 1)
    message(FATAL_ERROR "gRPC, Protobuf and Abseil were found in different installations:${_found}\n"
            "They must all come from one prefix. Delete ${CMAKE_BINARY_DIR}/CMakeCache.txt and CMakeFiles, then "
            "configure again with -DCMAKE_PREFIX_PATH=<your gRPC prefix> (see BUILDING.md).")
  endif()
endif()

# Require C++23 (overrides the C++17 default set in common.cmake)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# OpenTofu provider protocol (tfplugin5)
get_filename_component(proto_abs_path "proto/tfplugin5.proto" ABSOLUTE)
get_filename_component(proto_path "${proto_abs_path}" PATH)
get_filename_component(proto_stem "${proto_abs_path}" NAME_WE)

# Generated sources
set(proto_srcs "${CMAKE_CURRENT_BINARY_DIR}/${proto_stem}.pb.cc")
set(proto_hdrs "${CMAKE_CURRENT_BINARY_DIR}/${proto_stem}.pb.h")
set(grpc_srcs "${CMAKE_CURRENT_BINARY_DIR}/${proto_stem}.grpc.pb.cc")
set(grpc_hdrs "${CMAKE_CURRENT_BINARY_DIR}/${proto_stem}.grpc.pb.h")
add_custom_command(
      OUTPUT "${proto_srcs}" "${proto_hdrs}" "${grpc_srcs}" "${grpc_hdrs}"
      COMMAND ${_PROTOBUF_PROTOC}
      ARGS --grpc_out "${CMAKE_CURRENT_BINARY_DIR}"
        --cpp_out "${CMAKE_CURRENT_BINARY_DIR}"
        -I "${proto_path}"
        --plugin=protoc-gen-grpc="${_GRPC_CPP_PLUGIN_EXECUTABLE}"
        "${proto_abs_path}"
      DEPENDS "${proto_abs_path}")

# go-plugin controller service (lets Core ask the plugin to shut down)
set(ctrl_srcs "${CMAKE_CURRENT_BINARY_DIR}/grpc_controller.pb.cc" "${CMAKE_CURRENT_BINARY_DIR}/grpc_controller.grpc.pb.cc")
set(ctrl_hdrs "${CMAKE_CURRENT_BINARY_DIR}/grpc_controller.pb.h" "${CMAKE_CURRENT_BINARY_DIR}/grpc_controller.grpc.pb.h")
get_filename_component(ctrl_abs_path "proto/grpc_controller.proto" ABSOLUTE)
add_custom_command(
      OUTPUT ${ctrl_srcs} ${ctrl_hdrs}
      COMMAND ${_PROTOBUF_PROTOC}
      ARGS --grpc_out "${CMAKE_CURRENT_BINARY_DIR}"
        --cpp_out "${CMAKE_CURRENT_BINARY_DIR}"
        -I "${proto_path}"
        --plugin=protoc-gen-grpc="${_GRPC_CPP_PLUGIN_EXECUTABLE}"
        "${ctrl_abs_path}"
      DEPENDS "${ctrl_abs_path}")

# go-plugin stdio service (stream of the plugin's stdout/stderr to Core)
set(stdio_srcs "${CMAKE_CURRENT_BINARY_DIR}/grpc_stdio.pb.cc" "${CMAKE_CURRENT_BINARY_DIR}/grpc_stdio.grpc.pb.cc")
set(stdio_hdrs "${CMAKE_CURRENT_BINARY_DIR}/grpc_stdio.pb.h" "${CMAKE_CURRENT_BINARY_DIR}/grpc_stdio.grpc.pb.h")
get_filename_component(stdio_abs_path "proto/grpc_stdio.proto" ABSOLUTE)
add_custom_command(
      OUTPUT ${stdio_srcs} ${stdio_hdrs}
      COMMAND ${_PROTOBUF_PROTOC}
      ARGS --grpc_out "${CMAKE_CURRENT_BINARY_DIR}"
        --cpp_out "${CMAKE_CURRENT_BINARY_DIR}"
        -I "${proto_path}"
        --plugin=protoc-gen-grpc="${_GRPC_CPP_PLUGIN_EXECUTABLE}"
        "${stdio_abs_path}"
      DEPENDS "${stdio_abs_path}")

# go-plugin broker service (stream for negotiating extra connections)
set(broker_srcs "${CMAKE_CURRENT_BINARY_DIR}/grpc_broker.pb.cc" "${CMAKE_CURRENT_BINARY_DIR}/grpc_broker.grpc.pb.cc")
set(broker_hdrs "${CMAKE_CURRENT_BINARY_DIR}/grpc_broker.pb.h" "${CMAKE_CURRENT_BINARY_DIR}/grpc_broker.grpc.pb.h")
get_filename_component(broker_abs_path "proto/grpc_broker.proto" ABSOLUTE)
add_custom_command(

      ARGS --grpc_out "${CMAKE_CURRENT_BINARY_DIR}"

        -I "${proto_path}"

# Include generated *.pb.h files
include_directories("${CMAKE_CURRENT_BINARY_DIR}")

# plugin_grpc_proto
add_library(plugin_grpc_proto
  ${ctrl_srcs}
  ${ctrl_hdrs}
  ${stdio_srcs}
  ${stdio_hdrs}
  ${broker_srcs}
  ${broker_hdrs}
  ${grpc_srcs}
  ${grpc_hdrs}
  ${proto_srcs}
  ${proto_hdrs})
target_link_libraries(plugin_grpc_proto
  absl::check
  ${_REFLECTION}
  ${_GRPC_GRPCPP}
  ${_PROTOBUF_LIBPROTOBUF})

  ## Tartget name (executable file) is same as the project name (tofu-plugin-cpp)
  set (plugin_exe_name "terraform-provider-workaround")
  add_executable(${plugin_exe_name} "src/tofu-plugin.cpp" "src/server-cert.cpp")
  target_link_libraries(${plugin_exe_name}
    plugin_grpc_proto
    absl::check
    absl::flags
    absl::flags_parse
    absl::log
    absl::log_initialize
    ${_REFLECTION}
    ${_GRPC_GRPCPP}
    ${_PROTOBUF_LIBPROTOBUF})
